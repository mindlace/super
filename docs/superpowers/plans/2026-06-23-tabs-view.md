# Tabs-like view Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the resizable left sidebar with a horizontal tab strip across the top; each spec is a tab (title · caption · per-plan links), the editor fills the full width below.

**Architecture:** A pure helper `buildTabs(groups, selectedPath)` derives the tab view-model (active tab, spec/plan mode, per-plan link labels and boxed flag) from the single existing `selectedPath` state — no new persisted state. A thin presentational `TabStrip.svelte` renders that model and reuses the existing `selectSpec` callback. `+page.svelte` switches from a horizontal sidebar+main split to a vertical strip-over-content column and drops the sidebar resize logic. `pairAndGroup` and `captionFor` are unchanged.

**Tech Stack:** SvelteKit 2, Svelte 5 runes, TypeScript, Tailwind CSS 4, Vitest.

## Global Constraints

- Svelte 5 runes (`$state`, `$derived`, `$props`), 2-space indent, single quotes, 100 char width.
- Tests are colocated `*.test.ts`, pure-unit (no FS/DOM mocking) — mirror `src/lib/caption.test.ts`.
- Spec/plan H1 titles are never prefixed with the project name.
- `pairAndGroup`, `captionFor`, `normalizedSlug`, recency filtering, discovery, watcher, SSE, all `/api/*` and `/open` endpoints, and `sr:selectedPath` persistence stay unchanged.

### Task 1: `buildTabs` view-model helper

**Files:**

**Interfaces:**
- Consumes: `DisplayGroup` from `src/lib/pairing.ts` (`{ parent: SpecEntry; children: SpecEntry[]; recency: number }`); `captionFor(group: DisplayGroup): string` from `src/lib/caption.ts`.
- Produces:
  - `planLabel(index: number): string` — `0 → 'plan'`, `n → 'plan ' + (n+1)`.
  - `interface PlanLink { label: string; path: string; boxed: boolean }`
  - `interface Tab { title: string; caption: string; path: string; active: boolean; plans: PlanLink[] }`
  - `buildTabs(groups: DisplayGroup[], selectedPath: string | null): Tab[]` — one `Tab` per group, in input order. `tab.path` is `group.parent.path` (what clicking the title opens). `plans` is one `PlanLink` per child in input order; `boxed` is true when `selectedPath === child.path`. `active` is true when `selectedPath === group.parent.path` OR any plan is boxed.

```ts
import { describe, it, expect } from 'vitest';
import { buildTabs, planLabel } from './tabs';
import type { DisplayGroup } from './pairing';
import type { SpecEntry } from '$lib/types';

function entry(part: Partial<SpecEntry> & { filename: string }): SpecEntry {
  return {
    path: '/p/' + (part.project ?? 'proj') + '/' + part.filename,
    filename: part.filename,
    title: part.title ?? part.filename,
    project: part.project ?? 'proj',
    worktree: part.worktree ?? null,
    type: part.type ?? 'spec',
    mtime: part.recency ?? 1000,
    recency: part.recency ?? 1000,
    onMain: part.onMain
  };
}

function group(parent: SpecEntry, children: SpecEntry[] = []): DisplayGroup {
  return { parent, children, recency: parent.recency };
}

describe('planLabel', () => {
  it('labels the first plan "plan" and later plans by position', () => {
    expect(planLabel(0)).toBe('plan');
    expect(planLabel(1)).toBe('plan 2');
    expect(planLabel(2)).toBe('plan 3');
  });
});

describe('buildTabs', () => {
  it('returns one tab per group in input order with the captionFor caption', () => {
    const g1 = group(entry({ filename: 'a-design.md', title: 'Alpha', worktree: 'wtA' }));
    const g2 = group(entry({ filename: 'b-design.md', title: 'Beta', worktree: 'wtB' }));
    const tabs = buildTabs([g1, g2], null);
    expect(tabs.map((t) => t.title)).toEqual(['Alpha', 'Beta']);
    expect(tabs[0].caption).toBe('wtA · proj');
    expect(tabs[0].path).toBe(g1.parent.path);
  });

  it('marks no tab active and no plan boxed when nothing is selected', () => {
    const child = entry({ filename: 'p.md', type: 'plan' });
    const g = group(entry({ filename: 's-design.md' }), [child]);
    const [tab] = buildTabs([g], null);
    expect(tab.active).toBe(false);
    expect(tab.plans.map((p) => p.boxed)).toEqual([false]);
  });

  it('activates the tab in spec mode when the parent path is selected', () => {
    const child = entry({ filename: 'p.md', type: 'plan' });
    const parent = entry({ filename: 's-design.md' });
    const [tab] = buildTabs([group(parent, [child])], parent.path);
    expect(tab.active).toBe(true);
    expect(tab.plans.every((p) => !p.boxed)).toBe(true);
  });

  it('activates the parent tab and boxes the matching plan when a plan is selected', () => {
    const newest = entry({ filename: 'p-new.md', type: 'plan' });
    const older = entry({ filename: 'p-old.md', type: 'plan' });
    const parent = entry({ filename: 's-design.md' });
    // children arrive newest-first, as pairAndGroup sorts them
    const [tab] = buildTabs([group(parent, [newest, older])], older.path);
    expect(tab.active).toBe(true);
    expect(tab.plans).toEqual([
      { label: 'plan', path: newest.path, boxed: false },
      { label: 'plan 2', path: older.path, boxed: true }
    ]);
  });

  it('renders a standalone plan as a tab with no plan links', () => {
    const standalone = group(entry({ filename: 'loose-plan.md', type: 'plan', title: 'Loose' }));
    const [tab] = buildTabs([standalone], null);
    expect(tab.title).toBe('Loose');
    expect(tab.plans).toEqual([]);
  });

  it('returns an empty array for no groups', () => {
    expect(buildTabs([], null)).toEqual([]);
  });
});
```

- [ ] **Step 3: Write the implementation**

Create `src/lib/tabs.ts`:

```ts
import { captionFor } from './caption';
import type { DisplayGroup } from './pairing';

export interface PlanLink {
  label: string;
  path: string;
  boxed: boolean;
}

export interface Tab {
  title: string;
  caption: string;
  path: string;
  active: boolean;
  plans: PlanLink[];
}

export function planLabel(index: number): string {
  return index === 0 ? 'plan' : `plan ${index + 1}`;
}

export function buildTabs(groups: DisplayGroup[], selectedPath: string | null): Tab[] {
  return groups.map((g) => {
    const plans: PlanLink[] = g.children.map((c, i) => ({
      label: planLabel(i),
      path: c.path,
      boxed: selectedPath === c.path
    }));
    const active = selectedPath === g.parent.path || plans.some((p) => p.boxed);
    return {
      title: g.parent.title,
      caption: captionFor(g),
      path: g.parent.path,
      active,
      plans
    };
  });
}
```

```svelte
<script lang="ts">
  import type { DisplayGroup } from '$lib/pairing';
  import { buildTabs } from '$lib/tabs';

  let {
    groups,
    selectedPath,
    onSelect
  }: {
    groups: DisplayGroup[];
    selectedPath: string | null;
    onSelect: (path: string) => void;
  } = $props();

  const tabs = $derived(buildTabs(groups, selectedPath));
</script>

<nav class="flex shrink-0 gap-1.5 overflow-x-auto bg-neutral-950 px-2 pt-2 text-sm">
  {#each tabs as tab (tab.path)}
    <div
      class="min-w-[12rem] max-w-[15rem] shrink-0 rounded-t-lg border border-b-0 px-3 py-2
        {tab.active ? 'border-neutral-700 bg-neutral-900' : 'border-neutral-800 bg-neutral-950'}"
    >
      <button
        type="button"
        class="block w-full break-words text-left leading-snug
          {tab.active ? 'text-blue-200' : 'text-neutral-300'}"
        onclick={() => onSelect(tab.path)}
      >{tab.title}</button>
      <span class="mt-1 block break-words text-xs text-neutral-500">{tab.caption}</span>
      {#if tab.plans.length > 0}
        <div class="mt-1.5 flex flex-wrap gap-1.5">
          {#each tab.plans as plan (plan.path)}
            <button
              type="button"
              class="text-xs text-blue-400 hover:text-blue-300
                {plan.boxed
                  ? 'rounded border border-blue-500 px-1.5 text-blue-300'
                  : 'underline'}"
              onclick={() => onSelect(plan.path)}
            >{plan.label}</button>
          {/each}
        </div>
      {/if}
    </div>
  {/each}
</nav>
```

Note: the strip background is `neutral-950`, the active tab's background is `neutral-900` (matching `<main>`) with `border-b-0`, so the active tab reads flush into the content below.

```svelte
  let sidebarWidth = $state(DEFAULT_W);

  function startResize(e: PointerEvent) {
    resizing = true;
    (e.currentTarget as HTMLElement).setPointerCapture(e.pointerId);

    if (!resizing) return;

    if (!resizing) return;
```
